strings.HasPrefix $url.Path "/" -}} {{- $cleanAbs := strings.TrimPrefix "/" (path.Clean $url.Path) -}} {{- if findRE $excludedPathPattern $cleanAbs -}} {{- errorf "%s links to excluded documentation
